   223→function formatNumber(n: number): string {
   224→  return new Intl.NumberFormat('fr-FR').format(n)
   225→}
   226→
   227→function formatDate(d: string | null | undefined): string {
   228→  if (!d) return '—'
   229→  return new Intl.DateTimeFormat('fr-FR', { day: '2-digit', month: 'short', year: 'numeric' }).format(new Date(d))
   230→}
   231→
   232→function formatDateTime(d: string | null | undefined): string {
   233→  if (!d) return '—'
   234→  return new Intl.DateTimeFormat('fr-FR', { day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' }).format(new Date(d))
   235→}
   236→
   237→function getScoreColor(score: number): string {
   238→  if (score >= 60) return 'text-emerald-600'
   239→  if (score >= 30) return 'text-amber-600'
   240→  return 'text-red-600'
   241→}
   242→
   243→function getScoreBg(score: number): string {
   244→  if (score >= 60) return 'bg-emerald-500'
   245→  if (score >= 30) return 'bg-amber-500'
   246→  return 'bg-red-500'
   247→}

   628→function HotelsPage({ toast }: { toast: ReturnType<typeof useToast>['toast'] }) {

   645→  const fetchHotels = useCallback(async () => {

   647→    try {

   650→      params.set('limit', limit.toString())
   651→      params.set('sortBy', sortBy)

   655→      if (statusFilter !== 'all') params.set('statusDigital', statusFilter)
   656→      if (priorityFilter !== 'all') params.set('priority', priorityFilter)

   660→        const data = await res.json()
   661→        setHotels(data.hotels)
   662→        setTotalPages(data.pagination.totalPages)
   663→        setTotal(data.pagination.total)
