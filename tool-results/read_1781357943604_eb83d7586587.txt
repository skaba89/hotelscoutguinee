   449→function MenuReservationPage({ toast, onNavigate }: { toast: ReturnType<typeof useToast>['toast']; onNavigate: (p: PageType) => void }) {
   450→  const [hotels, setHotels] = useState<Hotel[]>([])
   451→  const [stats, setStats] = useState<Stats | null>(null)
   452→  const [loading, setLoading] = useState(true)
   453→  const [search, setSearch] = useState('')
   454→  const [selectedRegion, setSelectedRegion] = useState('all')
   455→  const [selectedCategory, setSelectedCategory] = useState('all')
   456→  const [selectedHotel, setSelectedHotel] = useState<Hotel | null>(null)
   457→  const [showReservation, setShowReservation] = useState(false)
   458→  const [reservationForm, setReservationForm] = useState({
   459→    guestName: '', guestEmail: '', guestPhone: '',
   460→    checkIn: '', checkOut: '', guests: 1, roomType: 'standard',
   461→    specialRequests: '',
   462→  })
   463→  const [creating, setCreating] = useState(false)
   464→  const [reservations, setReservations] = useState<Reservation[]>([])
   465→  const [showReservations, setShowReservations] = useState(false)
   466→  const [activeReservation, setActiveReservation] = useState<Reservation | null>(null)
   467→  const [showPlanning, setShowPlanning] = useState(false)
   468→  const [newStep, setNewStep] = useState({ step: '', label: '' })
   469→  const [showAddStep, setShowAddStep] = useState(false)
   470→  const [updatingStep, setUpdatingStep] = useState<string | null>(null)
   471→
   472→  const fetchData = useCallback(async () => {
   473→    setLoading(true)
   474→    try {
   475→      const [hotelsRes, statsRes] = await Promise.all([
   476→        fetch('/api/hotels?limit=200&sortBy=score&sortOrder=desc'),
   477→        fetch('/api/stats'),
   478→      ])
   479→      if (hotelsRes.ok) {
   480→        const data = await hotelsRes.json()
   481→        setHotels(data.hotels)
   482→      }
   483→      if (statsRes.ok) {
   484→        setStats(await statsRes.json())
   485→      }
   486→    } catch {
   487→      toast({ title: 'Erreur', description: 'Impossible de charger les données', variant: 'destructive' })
   488→    } finally {
   489→      setLoading(false)
   490→    }
   491→  }, [toast])
   492→
   493→  useEffect(() => { fetchData() }, [fetchData])
   494→
   495→  const fetchReservations = useCallback(async () => {
   496→    try {
   497→      const res = await fetch('/api/reservations')
   498→      if (res.ok) {
   499→        const data = await res.json()
   500→        setReservations(data.reservations)
   501→      }
   502→    } catch {
   503→      /* ignore */
   504→    }
   505→  }, [])
   506→
   507→  useEffect(() => { fetchReservations() }, [fetchReservations])
   508→
   509→  // Derived data
   510→  const regions = stats ? Object.keys(stats.byRegion).sort() : []
   511→  const categories = ['all', ...new Set(hotels.map(h => h.stars > 0 ? `${h.stars} étoiles` : 'Non classé'))].filter(Boolean)
   512→
   513→  const filteredHotels = hotels.filter(h => {
   514→    if (search && !h.name.toLowerCase().includes(search.toLowerCase()) && !h.city.toLowerCase().includes(search.toLowerCase())) return false
   515→    if (selectedRegion !== 'all' && h.region !== selectedRegion) return false
   516→    if (selectedCategory !== 'all') {
   517→      if (selectedCategory === 'Non classé' && h.stars > 0) return false
   518→      if (selectedCategory !== 'Non classé' && h.stars !== parseInt(selectedCategory)) return false
   519→    }
   520→    return true
   521→  })
   522→
   523→  const handleCreateReservation = async () => {
   524→    if (!selectedHotel || !reservationForm.guestName || !reservationForm.checkIn || !reservationForm.checkOut) {
   525→      toast({ title: 'Champs manquants', description: 'Remplissez le nom, les dates et sélectionnez un hôtel', variant: 'destructive' })
   526→      return
   527→    }
   528→    setCreating(true)
   529→    try {
   530→      const res = await fetch('/api/reservations', {
   531→        method: 'POST',
   532→        headers: { 'Content-Type': 'application/json' },
   533→        body: JSON.stringify({
   534→          hotelId: selectedHotel.id,
   535→          ...reservationForm,
   536→        }),
   537→      })
   538→      if (res.ok) {
   539→        const data = await res.json()
   540→        toast({ title: 'Réservation créée', description: `Confirmation: ${data.reservation.confirmationCode}` })
   541→        setShowReservation(false)
   542→        setActiveReservation(data.reservation)
   543→        setShowPlanning(true)
   544→        fetchReservations()
   545→        setReservationForm({ guestName: '', guestEmail: '', guestPhone: '', checkIn: '', checkOut: '', guests: 1, roomType: 'standard', specialRequests: '' })
   546→      } else {
   547→        const data = await res.json()
   548→        toast({ title: 'Erreur', description: data.error || 'Impossible de créer la réservation', variant: 'destructive' })
   549→      }
   550→    } catch {
   551→      toast({ title: 'Erreur', description: 'Connexion impossible', variant: 'destructive' })
   552→    } finally {
   553→      setCreating(false)
   554→    }
   555→  }
   556→
   557→  const handleUpdateStep = async (stepId: string, status: string) => {
   558→    setUpdatingStep(stepId)
   559→    try {
   560→      const res = await fetch('/api/planning', {
   561→        method: 'PATCH',
   562→        headers: { 'Content-Type': 'application/json' },
   563→        body: JSON.stringify({ stepId, status }),
